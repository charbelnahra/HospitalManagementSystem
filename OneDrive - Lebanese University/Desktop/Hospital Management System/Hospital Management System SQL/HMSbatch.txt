@echo off
echo ======================================
echo Hospital Management System - Batch
echo ======================================

REM Folder containing SQL files
set LAB=C:\SQLData\SQL Server Management Studio 21

REM SQL Server connection (Windows Authentication)
set SERV=-S localhost -E

echo Creating database...
osql %SERV% -i "%LAB%\create-HMS.sql" ^
-o "%LAB%\HMS_createDB.log"

echo Creating tables...
osql %SERV% -i "%LAB%\create-HMStables.sql" ^
-o "%LAB%\HMS_tables.log"

echo Altering tables...
osql %SERV% -i "%LAB%\create-HMSaltertables.sql" ^
-o "%LAB%\HMS_altertables.log"

echo Creating indexes...
osql %SERV% -i "%LAB%\create-HMSindexes.sql" ^
-o "%LAB%\HMS_indexes.log"

echo Creating views...
osql %SERV% -i "%LAB%\create-HMS-views.sql" ^
-o "%LAB%\HMS_views.log"

echo Creating stored procedures...
osql %SERV% -i "%LAB%\create-HMSstoredprocedures.sql" ^
-o "%LAB%\HMS_storedprocedures.log"

echo Creating triggers...
osql %SERV% -i "%LAB%\create-HMS-triggers.sql" ^
-o "%LAB%\HMS_triggers.log"

echo Creating security objects...
osql %SERV% -i "%LAB%\create-HMSsecurity.sql" ^
-o "%LAB%\HMS_security.log"

echo Inserting initial data...
osql %SERV% -i "%LAB%\HMS-insertdata.sql" ^
-o "%LAB%\HMS_insertdata.log"

echo ======================================
echo Batch execution finished
echo ======================================
pause
